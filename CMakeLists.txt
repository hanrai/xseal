cmake_minimum_required(VERSION 3.20)

# Prefer Clang, but do not hardcode
if (NOT CMAKE_C_COMPILER AND NOT CMAKE_CXX_COMPILER)
    find_program(CLANG_C_COMPILER clang)
    find_program(CLANG_CXX_COMPILER clang++)
    if (CLANG_C_COMPILER AND CLANG_CXX_COMPILER)
        set(CMAKE_C_COMPILER "clang")
        set(CMAKE_CXX_COMPILER "clang++")
        message(STATUS "Auto-selected Clang as compiler.")
    endif()
endif()

# 🚀 Native Clang Mode (Linux)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(CMAKE_LINKER "ld.lld")
else()
    message(WARNING "You are not using Clang. Based on benchmarks, compiling XSeal with GCC can result in an over 30% performance degradation due to differences in AVX2 unrolling and ILP scheduling. For paper-grade performance, Clang is highly recommended.")
endif()

project(XSeal VERSION 1.0 LANGUAGES C CXX)

# 🚀 Export compilation database, let Language Server (clangd/cpptools) find headers
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# 🚀 Force global Release mode, completely prohibit Debug pollution (Updated Blueprint)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Forced Release Build" FORCE)

# [Modification Point]: Append -g to Release flags
# -g generates DWARF debug info for AMD uProf, it does not affect machine code quality and execution speed generated by -O3
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG -fno-omit-frame-pointer")
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG -fno-omit-frame-pointer")

add_compile_definitions(NDEBUG)
set(CMAKE_CXX_FLAGS_DEBUG "") 
set(CMAKE_C_FLAGS_DEBUG "")

# Set C/C++ standards
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Shield Microsoft C runtime security warnings (let fopen, sprintf work quietly)
add_compile_definitions(_REENTRANT NOMINMAX _ENABLE_EXTENDED_ALIGNED_STORAGE _SILENCE_CXX20_ALIGNED_STORAGE_DEPRECATION_WARNING _SILENCE_CXX23_ALIGNED_STORAGE_DEPRECATION_WARNING NDEBUG _CRT_SECURE_NO_WARNINGS)
# Add specific compilation options for Clang
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    # Enable CMPXCHG16B instruction support for 16-byte atomic operations
    add_compile_options(-mcx16)
    # Shield warnings caused by deprecation of aligned_storage in C++20/C++23 (legacy code of ips4o)
    add_compile_options(-Wno-deprecated-declarations)
    
    # 🚀 Enable PDB format support (DWARF)
    add_compile_options(-g)
    
    # Enable OpenMP parallel support (temporarily disabled for debug export)
    # add_compile_options(-fopenmp)
    
    # Force use lld
    set(CMAKE_EXE_LINKER_FLAGS "-fuse-ld=lld" CACHE STRING "" FORCE)
endif()

# Enable basic optimization
add_compile_options(-O3 -march=native)

# Define include directories
include_directories(
    ${CMAKE_SOURCE_DIR}/include
)

include(FetchContent)
set(FETCHCONTENT_QUIET OFF)

# Google Test, Benchmark & CLI11
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.16.0.zip)
FetchContent_Declare(googlebenchmark URL https://github.com/google/benchmark/archive/refs/tags/v1.9.0.zip)
FetchContent_Declare(
  cli11
  GIT_REPOSITORY https://github.com/CLIUtils/CLI11.git
  GIT_TAG v2.4.1
)
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest googlebenchmark cli11)

# --- Minimap2 for Comparison ---
FetchContent_Declare(
  minimap2
  GIT_REPOSITORY https://github.com/lh3/minimap2.git
  GIT_TAG v2.28
)
FetchContent_MakeAvailable(minimap2)
# Minimap2 sketch module
add_library(mm2_sketch STATIC ${minimap2_SOURCE_DIR}/sketch.c ${minimap2_SOURCE_DIR}/kthread.c ${minimap2_SOURCE_DIR}/kalloc.c)
target_include_directories(mm2_sketch PUBLIC ${minimap2_SOURCE_DIR})
target_compile_options(mm2_sketch PRIVATE -O3 -march=native)

find_package(Threads REQUIRED)

# ==============================================================================
# XSeal Interface Library
# ==============================================================================
add_library(xseal_lib INTERFACE)
target_include_directories(xseal_lib INTERFACE include)

# ==============================================================================
# XSeal Core Targets (from Profiler branch)
# ==============================================================================

# ==============================================================================
# XSeal Optimized Targets (from fix-verification-performance branch)
# ==============================================================================
add_executable(xseal_bench src/xseal_bench.cpp)
target_link_libraries(xseal_bench PRIVATE Threads::Threads uring CLI11::CLI11)

enable_testing()
add_test(
  NAME run_chunk_sweep
  COMMAND bash "${CMAKE_SOURCE_DIR}/experiments/microbenchmarks/run_chunk_sweep.sh"
          "$<TARGET_FILE:xseal_bench>")
set_tests_properties(
  run_chunk_sweep
  PROPERTIES LABELS "benchmark"
             TIMEOUT 3600)

add_executable(minimizer_test_scalar tests/verify/minimizer_test_scalar.cpp)
target_link_libraries(minimizer_test_scalar PRIVATE Threads::Threads)

add_executable(syncmer_test_scalar tests/verify/syncmer_test_scalar.cpp)
target_link_libraries(syncmer_test_scalar PRIVATE Threads::Threads)

add_executable(xseal_verify tests/verify/xseal_verify.cpp)
target_compile_options(xseal_verify PRIVATE -O3)

add_executable(minimizer_scalar_baseline tests/verify/minimizer_scalar_baseline.cpp)

add_executable(verify_streaming tests/verify/verify_streaming.cpp)
target_include_directories(verify_streaming PRIVATE ${CMAKE_SOURCE_DIR}/tests/verify)
target_link_libraries(verify_streaming PRIVATE Threads::Threads)
# ==============================================================================
# LTO (Link Time Optimization)
# ==============================================================================
include(CheckIPOSupported)
check_ipo_supported(RESULT lto_supported OUTPUT error)
if(lto_supported)
    message(STATUS "🚀 Detected compiler supports LTO, enabling global Link Time Optimization!")
    set_property(TARGET xseal_bench PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
else()
    message(WARNING "Current environment does not support LTO: ${error}")
endif()

# Add extreme performance flags for Xseal core targets
set(XSEAL_EXTRA_FLAGS -mavx2 -mbmi2 -mfma -ffast-math)
target_compile_options(xseal_bench PRIVATE ${XSEAL_EXTRA_FLAGS})
